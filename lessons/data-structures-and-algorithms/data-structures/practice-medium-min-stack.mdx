## Solution

![Values 5, 2, 7, 1 have running minima 5, 2, 2, 1. The minimum is read from the top of the second stack.](https://pub-d143c03df8bc45d5a4589bc793aa52e4.r2.dev/courses/03ce588b-f581-46ee-b98b-277331400e55/gallery/6cf2f7a8-7f7e-4d2b-bf6f-63f984c86624)

For every entry in the value stack, store the minimum at that depth in a second stack. On the first push, the minimum is the value itself. On later pushes, append the smaller of the new value and the previous minimum.

In the diagram, pushing `7` onto `[5, 2]` records another `2` in the minimum stack. Pushing `1` records `1`. A pop removes one entry from each stack, restoring the minimum at the previous depth. The same rule handles duplicate minimum values.

    ```js
    function runMinStack(operations, values) {
      class MinStack {
        constructor() {
          this.items = [];
          this.minimums = [];
        }
    
        push(value) {
          this.items.push(value);
          const previous = this.minimums[this.minimums.length - 1];
          this.minimums.push(previous === undefined ? value : Math.min(value, previous));
        }
    
        pop() {
          this.items.pop();
          this.minimums.pop();
        }
    
        top() {
          return this.items[this.items.length - 1];
        }
    
        getMin() {
          return this.minimums[this.minimums.length - 1];
        }
      }
    
      let stack;
      const output = [];
    
      for (let index = 0; index < operations.length; index++) {
        const operation = operations[index];
    
        if (operation === "MinStack") {
          stack = new MinStack();
          output.push(null);
        } else if (operation === "push") {
          stack.push(values[index][0]);
          output.push(null);
        } else if (operation === "pop") {
          stack.pop();
          output.push(null);
        } else if (operation === "top") {
          output.push(stack.top());
        } else {
          output.push(stack.getMin());
        }
      }
    
      return output;
    }
    ```

    def run_min_stack(operations, values):
        class MinStack:
            def __init__(self):
                self.items = []
                self.minimums = []
    
            def push(self, value):
                self.items.append(value)
    
                minimum = min(value, self.minimums[-1]) if self.minimums else value
                self.minimums.append(minimum)
    
            def pop(self):
                self.items.pop()
                self.minimums.pop()
    
            def top(self):
                return self.items[-1]
    
            def get_min(self):
                return self.minimums[-1]
    
        stack = None
        output = []
    
        for index, operation in enumerate(operations):
            if operation == "MinStack":
                stack = MinStack()
                output.append(None)
            elif operation == "push":
                stack.push(values[index][0])
                output.append(None)
            elif operation == "pop":
                stack.pop()
                output.append(None)
            elif operation == "top":
                output.append(stack.top())
            else:
                output.append(stack.get_min())
    
        return output

### Big O notation

| Measure | Complexity | Explanation |
|---|---:|---|
| Time | `O(n)` | Across `n` operations, dynamic-array pushes and pops take amortized `O(1)` time each. `getMin` and `top` each read a single slot. |
| Auxiliary space | `O(n)` | The value and minimum stacks can each hold up to n values. |
