## Solution

![In an eight-slot circular queue with front at 3 and rear at 7, the next insertion wraps to slot 0.](https://pub-d143c03df8bc45d5a4589bc793aa52e4.r2.dev/courses/03ce588b-f581-46ee-b98b-277331400e55/gallery/8ef41c1f-fd34-44e4-a6b4-44873f21749d)

The front index identifies the next item to remove. With capacity `k` and room available, insert at `(front + size) % k`. When at least one item is present, read the rear at `(front + size - 1) % k`. A dequeue advances `front` by one modulo `k` and reduces `size`.

With capacity 3, enqueue `1, 2, 3`, dequeue `1`, then enqueue `4`. The physical slots read `[4, 2, 3]`, but `front` is 1, so the queue reads `2, 3, 4`. Its front is `2` and its rear is `4`.

    ```js
    function runCircularQueue(operations, values) {
      class MyCircularQueue {
        constructor(capacity) {
          this.items = Array(capacity);
          this.capacity = capacity;
          this.frontIndex = 0;
          this.size = 0;
        }
    
        enQueue(value) {
          if (this.isFull()) return false;
          const rearIndex = (this.frontIndex + this.size) % this.capacity;
          this.items[rearIndex] = value;
          this.size++;
          return true;
        }
    
        deQueue() {
          if (this.isEmpty()) return false;
          this.frontIndex = (this.frontIndex + 1) % this.capacity;
          this.size--;
          return true;
        }
    
        Front() {
          return this.isEmpty() ? -1 : this.items[this.frontIndex];
        }
    
        Rear() {
          if (this.isEmpty()) return -1;
          const rearIndex =
            (this.frontIndex + this.size - 1) % this.capacity;
          return this.items[rearIndex];
        }
    
        isEmpty() {
          return this.size === 0;
        }
    
        isFull() {
          return this.size === this.capacity;
        }
      }
    
      let queue;
      const output = [];
    
      for (let index = 0; index < operations.length; index++) {
        const operation = operations[index];
    
        if (operation === "MyCircularQueue") {
          queue = new MyCircularQueue(values[index][0]);
          output.push(null);
        } else if (operation === "enQueue") {
          output.push(queue.enQueue(values[index][0]));
        } else {
          output.push(queue[operation]());
        }
      }
    
      return output;
    }
    ```

    ```python
    def run_circular_queue(operations, values):
        class MyCircularQueue:
            def __init__(self, capacity):
                self.items = [None] * capacity
                self.capacity = capacity
                self.front_index = 0
                self.size = 0
    
            def enQueue(self, value):
                if self.isFull():
                    return False
    
                rear_index = (self.front_index + self.size) % self.capacity
                self.items[rear_index] = value
                self.size += 1
                return True
    
            def deQueue(self):
                if self.isEmpty():
                    return False
    
                self.front_index = (self.front_index + 1) % self.capacity
                self.size -= 1
                return True
    
            def Front(self):
                return -1 if self.isEmpty() else self.items[self.front_index]
    
            def Rear(self):
                if self.isEmpty():
                    return -1
    
                rear_index = (
                    self.front_index + self.size - 1
                ) % self.capacity
                return self.items[rear_index]
    
            def isEmpty(self):
                return self.size == 0
    
            def isFull(self):
                return self.size == self.capacity
    
        queue = None
        output = []
    
        for index, operation in enumerate(operations):
            if operation == "MyCircularQueue":
                queue = MyCircularQueue(values[index][0])
                output.append(None)
            elif operation == "enQueue":
                output.append(queue.enQueue(values[index][0]))
            else:
                output.append(getattr(queue, operation)())
    
        return output
    ```

### Big O notation

| Measure | Complexity | Explanation |
|---|---:|---|
| Time | `O(k + n)` | Initialize `k` slots, then perform `n` constant-time queue operations. |
| Auxiliary space | `O(k)` | The fixed array stores exactly the queue capacity k. |
