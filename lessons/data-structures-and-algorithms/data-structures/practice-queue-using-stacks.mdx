## Solution

![Transferring 3, 2, 1 from input places 1 at the top of output, ready to leave first.](https://pub-d143c03df8bc45d5a4589bc793aa52e4.r2.dev/courses/03ce588b-f581-46ee-b98b-277331400e55/gallery/f33ed34c-dc7a-4d5c-bfc1-cd49dddcaf34)

Push new values onto an input stack. When a front value is needed and the output stack is empty, move every input value to the output stack.

For pushes `1, 2`, the first pop transfers both values, then returns `1`. Push `3` into the input stack. The output stack still has `2`, so the next peek must read `2`. Moving `3` now would put it ahead of an older item. Transfer again only when the output stack is empty.

  <Tab value="JavaScript">
    ```js
    function runQueueUsingStacks(operations, values) {
      class MyQueue {
        constructor() {
          this.input = [];
          this.output = [];
        }
    
        moveIfNeeded() {
          if (this.output.length > 0) return;
    
          while (this.input.length > 0) {
            this.output.push(this.input.pop());
          }
        }
    
        push(value) {
          this.input.push(value);
        }
    
        pop() {
          this.moveIfNeeded();
          return this.output.pop();
        }
    
        peek() {
          this.moveIfNeeded();
          return this.output[this.output.length - 1];
        }
    
        empty() {
          return this.input.length === 0 && this.output.length === 0;
        }
      }
    
      let queue;
      const output = [];
    
      for (let index = 0; index < operations.length; index++) {
        const operation = operations[index];
    
        if (operation === "MyQueue") {
          queue = new MyQueue();
          output.push(null);
        } else if (operation === "push") {
          queue.push(values[index][0]);
          output.push(null);
        } else {
          output.push(queue[operation]());
        }
      }
    
      return output;
    }
    ```

    ```python
    def run_queue_using_stacks(operations, values):
        class MyQueue:
            def __init__(self):
                self.input = []
                self.output = []
    
            def move_if_needed(self):
                if self.output:
                    return
    
                while self.input:
                    self.output.append(self.input.pop())
    
            def push(self, value):
                self.input.append(value)
    
            def pop(self):
                self.move_if_needed()
                return self.output.pop()
    
            def peek(self):
                self.move_if_needed()
                return self.output[-1]
    
            def empty(self):
                return not self.input and not self.output
    
        queue = None
        output = []
    
        for index, operation in enumerate(operations):
            if operation == "MyQueue":
                queue = MyQueue()
                output.append(None)
            elif operation == "push":
                queue.push(values[index][0])
                output.append(None)
            else:
                output.append(getattr(queue, operation)())
    
        return output
    ```

### Big O notation

| Measure | Complexity | Explanation |
|---|---:|---|
| Time | `O(n)` | Across `n` operations, each value transfers at most once. One pop or peek can take `O(n)`, but the amortized cost is `O(1)` per operation. |
| Auxiliary space | `O(n)` | The two stacks together store all queued values. |
